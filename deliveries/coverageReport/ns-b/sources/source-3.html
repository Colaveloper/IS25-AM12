     /**
      * Starts a background thread to continuously process user input.
      * The thread reads input from the console, validates it against the current screen's
      * requirements, and then invokes the appropriate action. If cheat codes are enabled,
      * input may be provided by the cheat system instead of user input.
      */
     private void startInputLoop() {
         new Thread(() -> {
             Scanner scanner = new Scanner(System.in);
             String input;
             while (true) {
                 if(CheatCodes.isCheatOn() && !CheatCodes.cheatEmpty()){
                     try {
                         input = CheatCodes.cheat();
                     } catch (InterruptedException e) {
                         throw new RuntimeException(e);
                     }
                 }
                 else {
                     input = scanner.nextLine();
                 }
 
                 // letting the user correct format errors
                 while (!currentScreen.isInputLegal(input)) {
                     if (input.equals("drop_connection")) controller.dropConnection();
                     if (input.equals("exit")) controller.quit();
                     else {
                         System.out.println("Invalid format, please check your input");
                     }
                     input = scanner.nextLine();
                 }
 
                 currentScreen.parseAndInvoke(input);
             }
         }).start();
     }
